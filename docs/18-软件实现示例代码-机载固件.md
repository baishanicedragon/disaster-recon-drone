## 18.2 公共定义 `hdrs_defs.h`

```c
/* hdrs_defs.h —— 示例定义，与 17.8 协议常量表一致 */
#ifndef HDRS_DEFS_H
#define HDRS_DEFS_H
#include <stdint.h>

/* ---------- 协议常量（17.8） ---------- */
#define COMM_CUT_ALT      20.0f   /* m  不可经 APP 改写 */
#define VS_THRESH         20.9f   /* m/s 1.1 Vs，09 章 */
#define H_DEPLOY_MIN      90.0f   /* m   09 章 */
#define CLEAR_H           150.0f  /* m   08 章走廊净空 */
#define LOITER_SEC        300u    /* s   02 章盘旋 5 min */
#define RETURN_MARGIN     1.15f   /*     08 章逆风余量 */
#define CLIMB_CUTOFF_H    150.0f  /* m   段 I 关机高度，07 章 */

/* ---------- 检查单阈值（17.1 / 16.7） ---------- */
#define CELL_MIN_V        3.50f   /* V   电芯下限 */
#define CELL_DIFF_MAX_V   0.20f   /* V   最大压差 */
#define WING_LOCK_CNT     4
#define PROP_CNT          4

/* ---------- 顶层状态（15.5.1） ---------- */
typedef enum {
    S_BOOT = 0, S_WIFI_PAIR, S_PREP, S_ARMED, S_CLIMB,
    S_ABORT_RECYCLE, S_RUN, S_DESCEND, S_LAND,
    S_CUTOFF_SMOKE, S_RECOVER_WIFI, S_RECOVER_TF, S_END
} fms_state_t;

/* ---------- 飞行段（09 章） ---------- */
typedef enum { SEG_I = 1, SEG_II, SEG_III, SEG_IV, SEG_RTN } seg_t;

/* ---------- 互锁编号 IL-1~IL-6（07 章保留） ---------- */
typedef enum {
    IL_1_BATT = 0,   /* 电压/压差 */
    IL_2_WINGLOCK,   /* 翼锁销 ×4 */
    IL_3_PROP,       /* 桨顺桨 ×4 */
    IL_4_WIRE,       /* 配电回路导通 */
    IL_5_AIRSPEED,   /* 空速 ≥ VS_THRESH */
    IL_6_HEIGHT      /* 高度 ≥ H_DEPLOY_MIN */
} il_id_t;

/* ---------- 检查单状态字（回 APP 用，16.7） ---------- */
typedef struct {
    uint8_t  volt_ok;                    /* IL-1 */
    uint8_t  wing_lock[WING_LOCK_CNT];   /* IL-2，1=已锁 */
    uint8_t  prop[PROP_CNT];             /* IL-3，1=已顺桨 */
    uint8_t  wire_ok;                    /* IL-4 */
    float    v_min, v_diff;              /* 诊断用 */
} chk_word_t;

/* ---------- 航点 / ROI（17.7） ---------- */
typedef struct { int32_t lat, lon; int16_t alt_agl; uint8_t spd; } wp_t;
typedef struct { int32_t lat, lon; } pt_t;

typedef struct {
    uint8_t  cls;            /* 见 16.5.1 类别表 */
    uint8_t  poly_cnt;
    pt_t     poly[8];
    char     label[24];
    uint8_t  priority;
    uint8_t  expect_cnt;     /* 未达此数 → MISSING 告警 */
} roi_t;

typedef struct {
    uint16_t wp_cnt;   wp_t wp[64];
    uint8_t  fence_cnt; pt_t fence[16];  uint16_t clear_h;
    pt_t     recover;  float marker_expect[3];  /* x, y, heading */
    uint8_t  roi_cnt;  roi_t roi[16];
    float    vs_thresh; uint16_t h_deploy_min;
    float    comm_cut_alt; uint16_t loiter_sec; float return_margin;
} mission_t;

/* ---------- WiFi 帧类型（17.3.2） ---------- */
typedef enum {
    F_HELLO = 0x01, F_READY = 0x02, F_MISSION = 0x03,
    F_ACK = 0x04,   F_NAK = 0x05,   F_CHK = 0x06,
    F_ARM = 0x07,   F_ABORT = 0x08, F_BEACON = 0x09, F_DATA = 0x0A
} frame_type_t;
```

## 18.3 顶层状态机 `fms.c`

> 10 ms 调度（17.2）。示例只给骨架与出口条件，段 II~IV 的控制律见 09 章 `deploy_ctrl`。

```c
/* fms.c —— 顶层状态机骨架（示例） */
static fms_state_t st = S_BOOT;
static mission_t   M;          /* 已校验的任务文件 */
static chk_word_t  CHK;
static uint32_t    t_state_ms; /* 进入当前状态的时刻 */
static uint8_t     mission_ok;

void fms_init(void){ st = S_BOOT; t_state_ms = millis(); }

static void go(fms_state_t s){ st = s; t_state_ms = millis(); }

void fms_step_10ms(void)
{
    switch (st) {

    case S_BOOT:
        /* 17.2.1 自检：IMU 零偏 / 电池 / 桨锁 / SD */
        if (boot_selftest() != 0) { led_set(LED_RED_SOLID); buzzer(3); break; }
        go(S_WIFI_PAIR);
        break;

    case S_WIFI_PAIR:
        led_set(LED_BLUE_SLOW);                       /* 慢闪蓝：待配对 */
        wifi_ap_start("HDRS1_", board_sn4());         /* AP: HDRS1_<sn4> */
        if (mission_ok) {                             /* 收齐且 SHA256 过 */
            wifi_ap_stop_tx_only_up();                /* 保留上行直到 PREP */
            go(S_PREP);
        } else if (millis() - t_state_ms > 5*60*1000UL) {
            go(S_BOOT);                               /* 17.2.2 超时回 BOOT */
        }
        break;

    case S_PREP:
        led_set(LED_BLUE_FAST);                       /* 快闪蓝：待检查单 */
        checklist_sample(&CHK);                       /* 18.5 */
        wifi_send_chk(&CHK);                          /* 周期回 APP */
        if (cmd_pending(F_ARM) && checklist_all_green(&CHK)) go(S_ARMED);
        /* 注意：检查单未全绿时 ARM 被忽略（16.7 软硬双保险） */
        break;

    case S_ARMED:
        led_set(LED_BLUE_SOLID);
        /* 最后 3 s 静止自检（IMU 稳 + 电压稳），09 章 */
        if (!armed_final_selfcheck()) { if (millis()-t_state_ms > 3000) go(S_PREP); break; }
        if (launch_cmd_or_button()) go(S_CLIMB);
        break;

    case S_CLIMB:
        led_set(LED_GREEN_BLINK);
        seg_i_climb_control();                        /* 09 章段 I 倒拖 */
        /* comm_cut 在 18.4 独立判：alt ≥ 20 m 即切断并转 RUN，此处不重复判 */
        if (cmd_pending(F_ABORT)) { go(S_ABORT_RECYCLE); break; }
        if (agl_m() >= CLIMB_CUTOFF_H) deploy_gate_eval(&M);  /* 段 II 门控 18.8 */
        break;

    case S_ABORT_RECYCLE:
        led_set(LED_YELLOW_BLINK);
        prop_feather();  descend_hold();              /* 就地降高 */
        if (on_ground()) { power_cut(); smoke_fire(1); go(S_RECOVER_WIFI); }
        break;

    case S_RUN:
        led_set(LED_GREEN_SOLID);
        /* 指令链已切断：任何下行（含 ABORT）在 wifi_link 层即被丢弃 */
        run_mission(&M);                              /* 段 II~IV + 返航 */
        if (seg_done(SEG_RTN)) go(S_DESCEND);
        break;

    case S_DESCEND:
        led_set(LED_GREEN_TO_RED);
        descend_profile();                            /* 06 章迫降剖面 */
        if (on_ground()) go(S_LAND);
        break;

    case S_LAND:
        led_set(LED_RED_BLINK);
        datalog_flush_and_close();                    /* 18.10 后备电完成写盘 */
        go(S_CUTOFF_SMOKE);
        break;

    case S_CUTOFF_SMOKE:
        power_cut();                                  /* 断主电，保留 RTC/SD */
        smoke_fire(1);                                /* GPIO 单发烟雾 */
        go(battery_alive() ? S_RECOVER_WIFI : S_RECOVER_TF);
        break;

    case S_RECOVER_WIFI:                              /* 活体：重启 AP 等回传 */
        led_set(LED_BLUE_SLOW);
        wifi_ap_start("HDRS1_", board_sn4());
        if (recover_push_done()) go(S_END);
        break;

    case S_RECOVER_TF:                                /* 死体：等拔卡，无动作 */
        led_set(LED_OFF);
        break;

    case S_END: default: break;
    }
}
```

## 18.4 `comm_cut.c`：> 20 m 通讯切断（**不可屏蔽**）

> 15.5.1 硬规则：`COMM_CUT_ALT` 不可改写；进入 RUN 后丢弃一切下行。
> 建议放在**独立的高优先级任务 / 中断上下文**，不依赖 `fms` 调度是否卡死。

```c
/* comm_cut.c —— 独立于 fms，建议 100 Hz 或更高 */
#include "hdrs_defs.h"

static volatile uint8_t downlink_open = 1;   /* 1=可收下行, 0=已切断 */

/* AGL = 气压高(MSL) − DEM 高程；GNSS 拒止时退纯气压，误差打标（17.4） */
float agl_m(void)
{
    float baro_msl = baro_read_msl(&baro[0]);          /* DPS310 主，双余量 */
    float dem      = terrain_dem_elev_at(gnss_lat(), gnss_lon()); /* 08 章 DEM */
    float a = baro_msl - dem;
    if (!gnss_valid()) telemetry_mark("AGL_DEGRADED_BARO_ONLY");
    return a;
}

void comm_cut_task(void)
{
    /* 无论 fms 处于何态，只要越阈就切断——不提供任何"重新打开"的接口 */
    if (downlink_open && fms_state_get() == S_CLIMB && agl_m() >= COMM_CUT_ALT) {
        wifi_drop_downlink();        /* 关闭接收，ABORT 从此失效 */
        wifi_set_tx_only();          /* 仅保留可选 1 Hz beacon */
        downlink_open = 0;
        fms_force_state(S_RUN);      /* 转 RUN，段 II~IV 自主 */
        telemetry_mark("COMM_CUT@%.1fm", COMM_CUT_ALT);
    }
}

/* wifi_link 收包入口调用：切断后一切下行直接丢弃 */
uint8_t link_rx_accept(frame_type_t t)
{
    if (!downlink_open) return 0;                /* RUN 态：全部丢弃 */
    if (fms_state_get() != S_CLIMB && t == F_ABORT) return 0;  /* 仅 CLIMB 有效 */
    return 1;
}
```

## 18.5 `checklist.c`：IL-1~IL-4 状态字采集

```c
/* checklist.c —— 16.7 三步检查单的数据来源 */
#include "hdrs_defs.h"

void checklist_sample(chk_word_t* c)
{
    /* IL-1 电压：12S，取最低电芯与最大压差（INA226 + 从控） */
    batt_read(&c->v_min, &c->v_diff);
    c->volt_ok = (c->v_min >= CELL_MIN_V) && (c->v_diff <= CELL_DIFF_MAX_V);

    /* IL-2 翼锁销：4 片半翼，任一未锁即红 */
    c->wing_lock[0] = gpio_read(WING_LOCK_GPIO[0]);
    c->wing_lock[1] = gpio_read(WING_LOCK_GPIO[1]);
    c->wing_lock[2] = gpio_read(WING_LOCK_GPIO[2]);
    c->wing_lock[3] = gpio_read(WING_LOCK_GPIO[3]);

    /* IL-3 桨顺桨：4 桨，任一未顺桨即红 */
    for (int i = 0; i < PROP_CNT; i++) c->prop[i] = gpio_read(PROP_FEATHER_GPIO[i]);

    /* IL-4 导线导通：主供 + ESC 回路分压采样 */
    c->wire_ok = (adc_read(WIRE_ADC_CH) > WIRE_ADC_MIN);
}

uint8_t checklist_all_green(const chk_word_t* c)
{
    if (!c->volt_ok || !c->wire_ok) return 0;
    for (int i = 0; i < WING_LOCK_CNT; i++) if (!c->wing_lock[i]) return 0;
    for (int i = 0; i < PROP_CNT;     i++) if (!c->prop[i])      return 0;
    return 1;
}
```

## 18.6 `wifi_link.c`：帧编解码（17.3）

```c
/* wifi_link.c —— 帧格式 [0xAA][0x55][type][len u16][payload][crc16] */
#define FRAME_SOF0 0xAA
#define FRAME_SOF1 0x55

int wifi_send(frame_type_t t, const void* pl, uint16_t n)
{
    uint8_t buf[512];
    buf[0]=FRAME_SOF0; buf[1]=FRAME_SOF1; buf[2]=(uint8_t)t;
    buf[3]=(uint8_t)(n & 0xFF); buf[4]=(uint8_t)(n >> 8);
    if (n) memcpy(&buf[5], pl, n);
    uint16_t crc = crc16(buf, 5 + n);
    buf[5+n]=(uint8_t)(crc & 0xFF); buf[6+n]=(uint8_t)(crc >> 8);
    return uart_write(wifi_uart, buf, 7 + n);
}

/* 收包：先过 comm_cut 闸门，再分发 */
void wifi_rx_poll(void)
{
    frame_type_t t; uint8_t pl[512]; uint16_t n;
    while (frame_try_parse(wifi_uart, &t, pl, &n)) {
        if (!link_rx_accept(t)) continue;            /* 18.4 闸门：切断后丢弃 */
        switch (t) {
        case F_HELLO:
            wifi_send(F_READY, &(ready_t){.hw_ver=HW_VER, .status=S_WIFI_PAIR},
                      sizeof(ready_t));
            break;
        case F_MISSION:
            if (mission_rx_parse(pl, n, &M) == 0) {  /* 18.7，含 SHA256 */
                mission_ok = 1;
                wifi_send(F_ACK, M.sha256, 32);
            } else {
                wifi_send(F_NAK, "BAD_MISSION", 11);
            }
            break;
        case F_ARM:   cmd_post(F_ARM);   break;
        case F_ABORT: cmd_post(F_ABORT); break;      /* 仅 CLIMB 且 <20 m 可达此处 */
        default: break;
        }
    }
}

/* PREP 期周期回状态字：CHK{volt, wing_lock[4], prop[4], wire} */
void wifi_send_chk(const chk_word_t* c)
{
    uint8_t pl[16];
    pl[0] = c->volt_ok;
    memcpy(&pl[1], c->wing_lock, 4);
    memcpy(&pl[5], c->prop,      4);
    pl[9] = c->wire_ok;
    wifi_send(F_CHK, pl, 10);
}

/* RUN 期可选单向遥测 1 Hz（17.3.2 BEACON） */
void wifi_beacon_1hz(seg_t seg, float alt, uint16_t eta_s, float wh_used)
{
    if (!downlink_open) {                              /* 只发不收 */
        beacon_t b = {.seg=seg, .alt=alt, .eta=eta_s, .wh=wh_used};
        wifi_send(F_BEACON, &b, sizeof(b));
    }
}
```

## 18.7 `mission_rx.c`：解析 `mission.bin`（17.7 格式）

```c
/* mission_rx.c —— 小端；坐标 int32 定标 1e7；含 SHA256 校验 */
static uint16_t rd_u16(const uint8_t* p){ return (uint16_t)(p[0] | (p[1]<<8)); }
static uint32_t rd_u32(const uint8_t* p){ return (uint32_t)p[0] | ((uint32_t)p[1]<<8)
                                               | ((uint32_t)p[2]<<16) | ((uint32_t)p[3]<<24); }
static int32_t  rd_i32(const uint8_t* p){ return (int32_t)rd_u32(p); }
static float    rd_f32(const uint8_t* p){ float f; memcpy(&f,p,4); return f; }

int mission_rx_parse(const uint8_t* buf, uint32_t len, mission_t* m)
{
    const uint8_t* p = buf;
    if (len < 16) return -1;
    if (memcmp(p, "HDRS", 4) != 0) return -2;      /* magic */
    p += 4;
    uint16_t ver = rd_u16(p); p += 2;              /* ver */
    uint32_t crc = rd_u32(p); p += 4;              /* crc32 */
    uint32_t sz  = rd_u32(p); p += 4;              /* size  */
    if (sz > len - 14) return -3;
    if (crc32(p, sz) != crc) return -4;            /* 传输校验 */
    if (sha256_check(p, sz, buf + len - 32) != 0) return -5;  /* 尾部 sha256 */

    /* [TRAJ] */
    m->wp_cnt = rd_u16(p); p += 2;
    for (uint16_t i = 0; i < m->wp_cnt && i < 64; i++) {
        m->wp[i].lat = rd_i32(p); p += 4;
        m->wp[i].lon = rd_i32(p); p += 4;
        m->wp[i].alt_agl = (int16_t)rd_u16(p); p += 2;
        m->wp[i].spd = *p++;
    }
    m->fence_cnt = *p++;
    for (uint8_t j = 0; j < m->fence_cnt && j < 16; j++) {
        m->fence[j].lat = rd_i32(p); p += 4;
        m->fence[j].lon = rd_i32(p); p += 4;
    }
    m->clear_h = rd_u16(p); p += 2;                /* 默认 150 m */
    m->recover.lat = rd_i32(p); p += 4;
    m->recover.lon = rd_i32(p); p += 4;
    for (int k = 0; k < 3; k++) { m->marker_expect[k] = rd_f32(p); p += 4; }

    /* [ROI] */
    m->roi_cnt = *p++;
    for (uint8_t k = 0; k < m->roi_cnt && k < 16; k++) {
        roi_t* r = &m->roi[k];
        r->cls = *p++;
        r->poly_cnt = *p++;
        for (uint8_t q = 0; q < r->poly_cnt && q < 8; q++) {
            r->poly[q].lat = rd_i32(p); p += 4;
            r->poly[q].lon = rd_i32(p); p += 4;
        }
        memcpy(r->label, p, 24); p += 24;
        r->priority   = *p++;
        r->expect_cnt = *p++;
    }

    /* [PARAMS] —— 注意：comm_cut_alt 即使被写入也以本地常量为准，见下 */
    m->vs_thresh      = rd_f32(p); p += 4;
    m->h_deploy_min   = rd_u16(p); p += 2;
    float got_cut     = rd_f32(p); p += 4;
    m->loiter_sec     = rd_u16(p); p += 2;
    m->return_margin  = rd_f32(p); p += 4;

    /* 安全边界：忽略文件中的 comm_cut_alt，强制本地不可改写常量 */
    m->comm_cut_alt = COMM_CUT_ALT;
    (void)got_cut;
    (void)ver;
    return 0;
}
```

> **要点**：`comm_cut_alt` **不接受任务文件下发值**，强制取本地 `COMM_CUT_ALT`。
> 这是 17.4 安全边界在代码层的落实，任何"方便调试"的开关都不允许绕开它。

## 18.8 段 II 撑开门控（IL-5 / IL-6，沿用 09 章）

```c
/* deploy_ctrl 门控判定：空速 ≥ VS_THRESH 且 高度 ≥ H_DEPLOY_MIN */
typedef enum { GATE_WAIT = 0, GATE_OPEN, GATE_ABORT_PROFILE } gate_t;

gate_t deploy_gate_eval(const mission_t* m)
{
    float v  = aspd_read();            /* MS4525DO */
    float h  = agl_m();
    float vs = (float)m->vs_thresh;      /* 20.9 */
    float hd = (float)m->h_deploy_min;   /* 90   */

    /* 硬件互锁优先：动压开关 + 微动开关为独立信号，不纯靠软件（07 章 7.2.2） */
    uint8_t hw_ok = gpio_read(AIRSPEED_DYN_SW) && gpio_read(H_DEPLOY_SW);

    if ((v >= vs) && (h >= hd) && hw_ok) { wings_deploy(); return GATE_OPEN; }

    /* 高度已用尽仍未满足 → 放弃改平，走机头朝下迫降备用剖面 */
    if (h < hd * 0.5f) { emergency_nose_down_profile(); return GATE_ABORT_PROFILE; }
    return GATE_WAIT;
}

/* 拉平完成判定（09 章段 4）：|θ| < 15° 且 空速 > 1.15 Vs */
uint8_t flare_complete(void)
{
    return (fabsf(attitude_pitch_deg()) < 15.0f) && (aspd_read() > 1.15f * VS_1G);
}
```

```c
/* detect.c —— 段 IV 盘旋期，视觉 MCU 侧；结果落 annotations.json */
typedef struct { uint8_t cls; float conf; float x,y,w,h; } det_t;

/* 类别表（09 章 9.3.2 ↔ 16.5.1 模板 cls 映射） */
static const char* CLS_NAME[] = {"person","vehicle","structure","cross_red","debris"};

void detect_frame(uint32_t frame_id, float lat, float lon, float alt,
                  const det_t* d, uint8_t n, const mission_t* m)
{
    /* 1) NMS 已在视觉 MCU 完成，此处只做 ROI 匹配与落盘 */
    uint8_t found[16] = {0};   /* 每个 ROI 本帧命中数 */

    for (uint8_t i = 0; i < n; i++) {
        int roi_idx = roi_match_by_geo(m, lat, lon, d[i].cls);  /* 命中哪个 ROI */
        if (roi_idx >= 0) {
            found[roi_idx]++;
            annot_write_det(frame_id, CLS_NAME[d[i].cls], d[i].conf,
                            d[i].x, d[i].y, d[i].w, d[i].h,
                            m->roi[roi_idx].label);
            overlay_box_on_jpeg(frame_id, &d[i]);   /* 圈注叠加到 JPEG */
        }
    }

    /* 2) 期望数未达 → 记 MISSING（17.5） */
    for (uint8_t k = 0; k < m->roi_cnt; k++) {
        if (roi_accum(k) < m->roi[k].expect_cnt)
            annot_mark_missing(m->roi[k].label, m->roi[k].expect_cnt, roi_accum(k));
    }
}
```

```c
/* datalog.c —— /datalog/<task_id>/ 目录；双卡镜像（07 章 7.5.3） */
void datalog_open(const char* task_id)
{
    char path[64]; snprintf(path, sizeof(path), "/datalog/%s", task_id);
    f_mkdir(path);
    f_open(&f_vid,  "/datalog/%s/video.mp4", FA_WRITE | FA_CREATE_ALWAYS);
    f_open(&f_ann,  "/datalog/%s/annotations.json", FA_WRITE | FA_CREATE_ALWAYS);
    f_open(&f_tel,  "/datalog/%s/telemetry.log", FA_WRITE | FA_CREATE_ALWAYS);
    f_open(&f_echo, "/datalog/%s/mission_echo.bin", FA_WRITE | FA_CREATE_ALWAYS);
}

/* 循环落盘：每 5 s 强制 sync，不允许长时间缓存（07 章三保险之一） */
void datalog_tick_1s(void)
{
    static uint32_t last_sync = 0;
    if (millis() - last_sync >= 5000) {
        f_sync(&f_vid); f_sync(&f_ann); f_sync(&f_tel);
        mirror_copy_to_sd2();          /* 镜像到第二张卡 */
        last_sync = millis();
    }
}

/* LAND 断电前：后备电池（2S 200 mAh）保证最后 200 ms 完成写盘 + 关文件系统 */
void datalog_flush_and_close(void)
{
    f_sync(&f_vid); f_sync(&f_ann); f_sync(&f_tel); f_sync(&f_echo);
    f_close(&f_vid); f_close(&f_ann); f_close(&f_tel); f_close(&f_echo);
    mirror_final_sync();
}
```

## 18.12 移植与验证提示

| 项 | 说明 |
|:---|:---|
| **接 ArduPilot** | 平飞段（段 III/IV）不要自己写：用 `AUTO` + 地形跟随（`TERRAIN_ENABLE=1`）；倒拖段与改平段作为**自定义状态机**挂 AUTO 之外（07 章 7.2.2）。 |
| **`FS_GCS_ENABLE=0`** | 必须置 0：本设计无 GCS 链路，禁止链路失效触发 RTL（07 章 7.2.1）。 |
| **采样率** | `fms` 10 ms；`comm_cut` 建议 ≥ 100 Hz 且独立于 `fms`；段 I 控制 500 Hz（09 章）。 |
| **不可屏蔽逻辑** | `comm_cut` 与 IL-5/IL-6 的硬件互锁信号（动压开关/微动开关）**不得**由软件单独判定。 |
| **验证顺序** | 先系留测倒拖 → 滑翔机投放测无动力改平 → 全剖面合练（07 章 7.2.2 三步走）。 |
| **未覆盖** | 本示例不含 EKF、控制律整定、失效保护细节；这些用 ArduPilot 现成能力。 |
